from bisect import bisect_left, bisect_right
from collections import defaultdict
from datetime import timedelta

WINDOW = timedelta(days=3)

def match(transactions, invoices):
    # exact key first: amount in cents; then a sorted date list per bucket
    by_amount = defaultdict(list)
    for inv in invoices:
        by_amount[inv.cents].append((inv.date, inv))
    for dates in by_amount.values():
        dates.sort(key=lambda d: d[0])

    for tx in transactions:
        bucket = by_amount.get(tx.cents, [])
        keys = [d for d, _ in bucket]   # TODO: precompute — this is O(k) per tx
        lo = bisect_left(keys, tx.date - WINDOW)
        hi = bisect_right(keys, tx.date + WINDOW)
